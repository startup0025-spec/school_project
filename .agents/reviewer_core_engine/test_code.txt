import { checkGeofenceAndSafety, getSonificationParams } from './core_engine/src/api';
import { getPlaces } from './core_engine/src/database/local_places';

async function run() {
  const places = await getPlaces();
  console.log('Total places:', places.length);
  for (const place of places) {
    console.log('--- Place:', place.name, 'waterType:', place.waterType, 'station:', place.waterStationName || 'none');
    const params = await getSonificationParams(place);
    console.log('Audio Params:', JSON.stringify(params, null, 2));

    const nearLevel = await checkGeofenceAndSafety(place.latitude, place.longitude);
    console.log('Near Safety Level:', nearLevel);
  }

  // Test far coordinate (Seoul City Hall)
  const farLevel = await checkGeofenceAndSafety(37.5665, 126.9780);
  console.log('Far Safety Level (Seoul):', farLevel);
}

run().catch(console.error);
